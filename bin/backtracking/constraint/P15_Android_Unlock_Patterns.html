P15 — Android Unlock Patterns

Medium • LC #351 • Skip rule via "middle" lookup

Pattern

Precompute skip[a][b] = c (the cell between a and b on the keypad, 0 if none). Move from a → b allowed iff visited[b]==false AND (skip[a][b]==0 OR visited[skip[a][b]]).

Java

private int[][] skip = new int[10][10];
public int numberOfPatterns(int m, int n) {
    skip[1][3] = skip[3][1] = 2;
    skip[1][7] = skip[7][1] = 4;
    skip[3][9] = skip[9][3] = 6;
    skip[7][9] = skip[9][7] = 8;
    skip[1][9] = skip[9][1] = skip[2][8] = skip[8][2] = skip[3][7] = skip[7][3] = skip[4][6] = skip[6][4] = 5;
    boolean[] vis = new boolean[10];
    int total = 0;
    for (int len = m; len <= n; len++) {
        total += 4 * dfs(1, vis, len - 1);
        total += 4 * dfs(2, vis, len - 1);
        total += dfs(5, vis, len - 1);
    }
    return total;
}
int dfs(int cur, boolean[] vis, int left) {
    if (left == 0) return 1;
    vis[cur] = true; int res = 0;
    for (int next = 1; next <= 9; next++)
        if (!vis[next] && (skip[cur][next] == 0 || vis[skip[cur][next]])) res += dfs(next, vis, left - 1);
    vis[cur] = false;
    return res;
}
Insight: Symmetry: counts for corners (×4) and edges (×4) are equal; computing one of each saves a factor of 4.
