// same[i] = #ways post i matches post i-1; diff[i] = #ways it differs.
int same = 0, diff = k;
for (int i = 2; i <= n; i++) {
    int newSame = diff;            // same as previous post
    int newDiff = (same + diff) * (k - 1);
    same = newSame; diff = newDiff;
}
return same + diff;
