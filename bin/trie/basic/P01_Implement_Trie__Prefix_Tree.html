P01 — Implement Trie

Medium • LC #208

Java

class Trie {
    static class N { N[] kids = new N[26]; boolean end; }
    private N root = new N();
    public void insert(String w) {
        N n = root;
        for (char c : w.toCharArray()) {
            int i = c - 'a';
            if (n.kids[i] == null) n.kids[i] = new N();
            n = n.kids[i];
        }
        n.end = true;
    }
    private N walk(String s) {
        N n = root;
        for (char c : s.toCharArray()) { n = n.kids[c-'a']; if (n == null) return null; }
        return n;
    }
    public boolean search(String w) { N n = walk(w); return n != null && n.end; }
    public boolean startsWith(String p) { return walk(p) != null; }
}
Insight: The 'end' flag distinguishes "is a word" from "is a prefix only."
