TrieBasic

Trie · Basic — Masterclass

A trie is a 26-ary tree (for lowercase letters) where each path from root to node spells a prefix. Insert, search, and prefix-walk operations are all O(L).

Template

class TrieNode {
    TrieNode[] kids = new TrieNode[26];
    boolean end;
}
void insert(String w) {
    TrieNode n = root;
    for (char c : w.toCharArray()) {
        if (n.kids[c-'a'] == null) n.kids[c-'a'] = new TrieNode();
        n = n.kids[c-'a'];
    }
    n.end = true;
}
boolean search(String w) {
    TrieNode n = walk(w);
    return n != null && n.end;
}
boolean startsWith(String p) { return walk(p) != null; }
