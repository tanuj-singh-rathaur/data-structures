P01 — Number of Islands

Medium • LC #200

Java

public int numIslands(char[][] g) {
    int n = 0;
    for (int r=0;r<g.length;r++) for (int c=0;c<g[0].length;c++)
        if (g[r][c]=='1') { n++; bfs(g, r, c); }
    return n;
}
void bfs(char[][] g, int sr, int sc) {
    Queue<int[]> q = new ArrayDeque<>();
    q.offer(new int[]{sr, sc});
    g[sr][sc] = '0';
    int[][] d = {{0,1},{0,-1},{1,0},{-1,0}};
    while (!q.isEmpty()) {
        int[] c = q.poll();
        for (int[] e : d) {
            int nr=c[0]+e[0], nc=c[1]+e[1];
            if (nr>=0 && nr<g.length && nc>=0 && nc<g[0].length && g[nr][nc]=='1') {
                g[nr][nc] = '0';
                q.offer(new int[]{nr, nc});
            }
        }
    }
}
Insight: Counting connected components — each BFS flood-fills one island.
