Graphs · Topological Sort — Masterclass

Linear ordering of nodes in a DAG such that every edge u→v has u before v. Two algorithms: Kahn (BFS, indegree-driven) and DFS-postorder.

Templates

// Kahn's algorithm (BFS)
int[] indeg = new int[n];
for (int u = 0; u < n; u++) for (int v : adj.get(u)) indeg[v]++;
Queue<Integer> q = new ArrayDeque<>();
for (int i = 0; i < n; i++) if (indeg[i] == 0) q.offer(i);
List<Integer> order = new ArrayList<>();
while (!q.isEmpty()) {
    int u = q.poll(); order.add(u);
    for (int v : adj.get(u)) if (--indeg[v] == 0) q.offer(v);
}
// order.size() < n ⇒ cycle exists

// DFS-based
void dfs(int u) {
    state[u] = 1;
    for (int v : adj.get(u)) if (state[v] == 0) dfs(v);
    state[u] = 2; stack.push(u);
}
// Final order = reverse(stack)
